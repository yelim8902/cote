Python
def insertion_sort(arr): for i in range(1, len(arr)): key = arr[i] j = i - 1 while j >= 0 and arr[j] > key: arr[j + 1] = arr[j] j -= 1 arr[j + 1] = key data = [5, 3, 4, 1, 2] insertion_sort(data) print("Sorted array:", data) # Sorted array: [1, 2, 3, 4, 5]

Java
public class InsertionSort { public static void insertionSort(int[] arr) { for (int i = 1; i < arr.length; i++) { int key = arr[i]; int j = i - 1; while (j >= 0 && arr[j] > key) { arr[j + 1] = arr[j]; j--; } arr[j + 1] = key; } } public static void main(String[] args) { int[] data = {5, 3, 4, 1, 2}; insertionSort(data); System.out.print("Sorted array: "); for (int num : data) { System.out.print(num + " "); } // Sorted array: 1 2 3 4 5 } }
