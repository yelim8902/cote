배열 연산(접근·삽입·삭제·순회)을 코드로 확인한 뒤, 오른쪽 칸에 따라 적어 보세요.

# Python: 리스트 = 동적 배열
arr = [10, 20, 30, 40]

# 접근 O(1)
x = arr[2]           # 30
arr[1] = 99          # 수정

# 순회 O(n)
for v in arr:
    print(v)

# 삽입 O(n) - insert(위치, 값)
arr.insert(2, 25)    # 인덱스 2에 25 삽입

# 삭제 O(n) - pop(위치) 또는 remove(값)
arr.pop(1)           # 인덱스 1 제거
arr.remove(30)       # 값 30 제거

# 정적처럼 쓰려면 크기 고정 후 인덱스로만 접근

# 사용 예
arr = [10, 20, 30, 40]
print(arr[2])        # 30
arr.insert(2, 25)    # [10, 20, 25, 30, 40]
arr.pop(1)           # [10, 25, 30, 40]
for v in arr:
    print(v, end=" ")  # 10 25 30 40
// Java: 정적 배열 / ArrayList(동적)

    System.out.println(list.get(i));

#include <stdio.h>
#include <stdlib.h>

/* ========== 정적 배열 ==========
 * 컴파일 시 크기가 고정. 인덱스로 직접 접근 → O(1) */
int arr[100];
arr[0] = 5;       /* 0번 인덱스에 쓰기 */
int x = arr[2];  /* 2번 인덱스 읽기 */

/* ========== 동적 배열 ==========
 * malloc으로 힙에 할당. 사용 후 free() 필요. 접근 역시 O(1) */
int n = 10;
int* dyn = (int*)malloc(n * sizeof(int));
dyn[0] = 1;  dyn[1] = 2;

/* ========== 중간 삽입 O(n) ==========
 * a: 배열 포인터, len: 현재 원소 개수(포인터로 갱신), cap: 최대 용량
 * idx 위치에 val 삽입 → idx 이후 원소들을 한 칸씩 뒤로 밀고, idx에 val 저장 */
void insert_at(int* a, int* len, int cap, int idx, int val) {
    if (*len >= cap) return;   /* 꽉 찼으면 삽입 불가 */
    for (int i = *len; i > idx; i--) a[i] = a[i-1];  /* 뒤에서부터 밀기 */
    a[idx] = val; (*len)++;
}

/* ========== 중간 삭제 O(n) ==========
 * idx 위치 원소 제거 → idx+1 이후 원소들을 한 칸씩 앞으로 당김 */
void delete_at(int* a, int* len, int idx) {
    for (int i = idx; i < *len-1; i++) a[i] = a[i+1];
    (*len)--;
}

/* 사용 예: len은 현재 길이를 담고 있어야 삽입/삭제 시 갱신됨 */
int a[10], len = 0, cap = 10;
insert_at(a, &len, cap, 0, 10);
insert_at(a, &len, cap, 1, 20);
insert_at(a, &len, cap, 2, 30);
delete_at(a, &len, 0);   /* a[] = {20, 30}, len=2 */
