def dfs(adj, start):
    vis = set()
    stack = [start]
    while stack:
        v = stack.pop()
        if v in vis: continue
        vis.add(v)
        for u in adj[v]:
            if u not in vis: stack.append(u)

def bfs(adj, start):
    from collections import deque
    vis, q = {start}, deque([start])
    while q:
        v = q.popleft()
        for u in adj[v]:
            if u not in vis: vis.add(u); q.append(u)
void dfs(List> adj, int start) {
    boolean[] vis = new boolean[adj.size()];
    Stack st = new Stack<>();
    st.push(start);
    while (!st.isEmpty()) {
        int v = st.pop();
        if (vis[v]) continue;
        vis[v] = true;
        for (int u : adj.get(v)) if (!vis[u]) st.push(u);
    }
}
void bfs(List> adj, int start) {
    boolean[] vis = new boolean[adj.size()];
    Queue q = new LinkedList<>();
    q.offer(start); vis[start] = true;
    while (!q.isEmpty()) {
        int v = q.poll();
        for (int u : adj.get(v)) if (!vis[u]) { vis[u]=true; q.offer(u); }
    }
}
void dfs(int adj[][], int n, int start) {
    int vis[MAX], st[MAX], top = 0;
    memset(vis, 0, sizeof vis);
    st[top++] = start;
    while (top) {
        int v = st[--top];
        if (vis[v]) continue;
        vis[v] = 1;
        for (int u = 0; u < n; u++)
            if (adj[v][u] && !vis[u]) st[top++] = u;
    }
}
