def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n - 1 - i):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
# 인접 원소 비교·교환, O(n²)

# 사용 예
# arr = [64, 34, 25, 12, 22]
# bubble_sort(arr)  # arr = [12, 22, 25, 34, 64]
void bubbleSort(int[] arr) {
    int n = arr.length;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (arr[j] > arr[j + 1]) {
                int t = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = t;
            }
}
// 사용 예: int[] arr = {64, 34, 25}; bubbleSort(arr);
#include <stdio.h>

/* 버블 정렬: 인접한 두 원소를 비교해 큰 것을 뒤로 보냄. 한 턴마다 맨 뒤에 최댓값이 확정.
 * 바깥 루프 i: n-1-i 까지만 보면 됨 (뒤쪽 i개는 이미 정렬됨). O(n²) */
void bubble_sort(int arr[], int n) {
    int i, j, t;
    for (i = 0; i < n; i++)
        for (j = 0; j < n - 1 - i; j++)
            if (arr[j] > arr[j+1]) {
                t = arr[j]; arr[j] = arr[j+1]; arr[j+1] = t;
            }
}
/* 사용 예: int arr[] = {64, 34, 25}; bubble_sort(arr, 3); */
