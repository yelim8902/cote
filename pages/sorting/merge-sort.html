def merge_sort(arr):
    if len(arr) <= 1: return arr
    mid = len(arr) // 2
    L = merge_sort(arr[:mid])
    R = merge_sort(arr[mid:])
    i = j = 0
    out = []
    while i < len(L) and j < len(R):
        if L[i] <= R[j]: out.append(L[i]); i += 1
        else: out.append(R[j]); j += 1
    out += L[i:] + R[j:]
    return out
# O(n log n), 안정 정렬

# 사용 예
# arr = [38, 27, 43, 3, 9, 82, 10]
# arr = merge_sort(arr)  # [3, 9, 10, 27, 38, 43, 82]
void mergeSort(int[] arr, int l, int r) {
    if (l >= r) return;
    int m = (l + r) / 2;
    mergeSort(arr, l, m);
    mergeSort(arr, m + 1, r);
    merge(arr, l, m, r);  // 두 구간 병합
}
// merge()에서 보조 배열로 정렬 후 복사
// 사용 예: mergeSort(arr, 0, arr.length - 1);
/* 병합 정렬: 구간 [l, r]을 반으로 나누어 각각 재귀 정렬한 뒤, merge(arr, l, m, r)로 두 구간을 합침.
 * merge: 보조 배열에 arr[l..m]과 arr[m+1..r]을 정렬하며 합쳐서 arr에 다시 복사. O(n log n), 안정 정렬 */
void merge_sort(int arr[], int l, int r) {
    if (l >= r) return;
    int m = (l + r) / 2;
    merge_sort(arr, l, m);
    merge_sort(arr, m + 1, r);
    merge(arr, l, m, r);
}
