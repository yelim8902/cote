선형 큐
앞(Front): -1 뒤(Rear): -1 크기: 0
원형 큐
앞(Front): -1 뒤(Rear): -1 크기: 0

두 큐를 초기화했습니다. 선형 큐는 앞 위치 고정, 원형 큐는 링 형태로 회전합니다.

코드로 학습하기

원형 큐(배열 기반) Enqueue/Dequeue를 front·rear로 구현한 코드입니다. 오른쪽 칸에 따라 적어 보세요.

class CircularQueue:
    def __init__(self, cap):
        self.cap = cap
        self.q = [None] * cap
        self.front = self.rear = 0

    def enqueue(self, x):
        if (self.rear + 1) % self.cap == self.front: return  # full
        self.rear = (self.rear + 1) % self.cap
        self.q[self.rear] = x

    def dequeue(self):
        if self.front == self.rear: return None  # empty
        self.front = (self.front + 1) % self.cap
        return self.q[self.front]

# 사용 예
cq = CircularQueue(5)
cq.enqueue(10)
cq.enqueue(20)
cq.enqueue(30)
print(cq.dequeue())   # 10
print(cq.dequeue())   # 20
class CircularQueue {
    int[] q; int cap, front = 0, rear = 0;
    CircularQueue(int cap) { this.cap = cap; q = new int[cap]; }
    void enqueue(int x) {
        if ((rear + 1) % cap == front) return;
        rear = (rear + 1) % cap;
        q[rear] = x;
    }
    int dequeue() {
        if (front == rear) return -1;
        front = (front + 1) % cap;
        return q[front];
    }
    public static void main(String[] args) {
        CircularQueue cq = new CircularQueue(5);
        cq.enqueue(10);
        cq.enqueue(20);
        cq.enqueue(30);
        System.out.println(cq.dequeue());  // 10
        System.out.println(cq.dequeue());  // 20
    }
}
#include <stdio.h>
#define MAX 100

/* 원형 큐: front와 rear를 0으로 시작. 한 칸 비워두고 (rear+1)%cap == front면 꽉 찬 것으로 봄.
 * cap은 큐 크기. 배열을 원형으로 쓰기 위해 인덱스를 % cap으로 계산. */
int q[MAX], front = 0, rear = 0, cap = MAX;

/* enqueue: 꽉 찼으면 ((rear+1)%cap == front) 그냥 return.
 * 아니면 rear를 다음 위치로 옮긴 뒤 (rear = (rear+1)%cap) 그 자리에 x 저장 */
void enqueue(int x) {
    if ((rear + 1) % cap == front) return;
    rear = (rear + 1) % cap;
    q[rear] = x;
}

/* dequeue: 비었으면 (front == rear) -1 반환.
 * 아니면 front를 다음 위치로 옮기고 (front = (front+1)%cap) 그 자리 값 반환 */
int dequeue() {
    if (front == rear) return -1;
    front = (front + 1) % cap;
    return q[front];
}

int main(void) {
    enqueue(10); enqueue(20); enqueue(30);
    printf("%d\n", dequeue());  /* 10 */
    printf("%d\n", dequeue());  /* 20 */
    return 0;
}
