Enqueue, Dequeue 연산을 코드로 확인한 뒤, 오른쪽 칸에 따라 적어 보세요.

from collections import deque
q = deque()
q.append(10)   # Enqueue O(1)
q.append(20)
q.append(30)
x = q.popleft()  # Dequeue O(1) → 10
# list로 하면: q.pop(0) 은 O(n)
import java.util.Queue;
import java.util.LinkedList;
Queue<Integer> q = new LinkedList<>();
q.offer(10);   // Enqueue O(1)
q.offer(20);
q.offer(30);
int x = q.poll();  // Dequeue O(1) → 10
// q.peek() 맨 앞 확인, q.isEmpty()
#include <stdio.h>
#define MAX 100
int q[MAX], front = 0, rear = -1;

void enqueue(int x) {
    if (rear < MAX - 1) q[++rear] = x;
}
int dequeue() {
    return front <= rear ? q[front++] : -1;
}
