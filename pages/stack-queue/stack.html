#include <stdio.h>
#define MAX 100

/* 스택: 배열 하나와 top 인덱스. top == -1이면 빈 스택. LIFO(Last In First Out) */
int stack[MAX], top = -1;

/* push: top을 먼저 증가시킨 뒤 그 위치에 x 저장. 꽉 찼으면 (top < MAX-1) 체크 */
void push(int x) {
    if (top < MAX - 1) stack[++top] = x;
}

/* pop: top이 유효하면 stack[top] 반환하고 top 감소. 빈 스택이면 -1 등 에러값 반환 */
int pop() {
    return top >= 0 ? stack[top--] : -1;
}

/* peek: 제거하지 않고 맨 위 값만 확인. top이 유효할 때만 stack[top] 반환 */
int peek() {
    return top >= 0 ? stack[top] : -1;
}

/* 사용 예 */
push(10); push(20); push(30);
printf("%d\n", pop());   /* 30 (마지막에 넣은 것 먼저 나옴) */
printf("%d\n", peek());  /* 20 (현재 맨 위) */
